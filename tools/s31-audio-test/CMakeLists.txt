cmake_minimum_required(VERSION 3.16)
# cspell:words Wextra Werror
project(s31_audio_test C)
find_package(Threads REQUIRED)
add_executable(s31_audio_test test_render.c)
target_include_directories(s31_audio_test PRIVATE stubs ../../components/s31_board/include)
set_property(TARGET s31_audio_test PROPERTY C_STANDARD 11)
target_link_libraries(s31_audio_test PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(s31_audio_test PRIVATE /W4 /experimental:c11atomics /UNDEBUG)
else()
    target_compile_options(s31_audio_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME s31_audio_lifecycle COMMAND s31_audio_test)
set_tests_properties(s31_audio_lifecycle PROPERTIES TIMEOUT 60)
